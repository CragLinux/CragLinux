# RPi firmware configuration (docs/04 §1: EEPROM -> start4.elf ->
# U-Boot -> boot.scr). Same chain as rpi4/revpi5: the Pi 5's EEPROM
# loads bcm2712-rpi-5-b.dtb from this partition, applies its fixups,
# and hands it to U-Boot — which passes it on to the kernel
# (boards/rpi5/uboot/boot.script.in). cmdline.txt is not used: the
# kernel command line comes from the A/B boot script.

# 64-bit mode, U-Boot as the firmware's "kernel"
arm_64bit=1
kernel=u-boot.bin

# Firmware+bootloader progress on the 3-pin debug UART (uart10);
# uart_2ndstage echoes it there before the kernel takes over. The
# kernel console is the GPIO14/15 mini-UART (ttyS0) + HDMI
# (boards/rpi5/board.toml [kernel].cmdline note).
enable_uart=1
uart_2ndstage=1

# Fast, headless-friendly defaults; HDMI stays functional.
disable_splash=1
avoid_warnings=1
